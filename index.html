async function analyzeWithGemini(apiKey, base64Image, mimeType) { const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`; const prompt = ` このパチンコデータアプリのスクリーンショットから、以下の13箇所の【現在の数値】を正確に抽出してJSON形式のみで返してください。 ハイフン「-」や未記録の項目は 0 としてください。 1. target_game: 「対象ゲーム数」の右横に書かれている数字 2. 4x4のボタングリッドの上から3段（計12個）のボタン中央にある白い大きな数字： - 1段目: - invest: 左端（お金のアイコンの上） - naki2: 左から2番目（②の上） - replay: 左から3番目（リプレイの上） - rep_v: 右端（リプ→Vの上） - 2段目: - pick: 左端（羽根拾の上） - v_hit: 左から2番目（V入賞の上） - sp: 左から3番目（SPの上） - sp_v: 右端（SP→Vの上） - 3段目: - kick: 左端（拾い→蹴りの上） - r10: 左から2番目（当大の上） - r5: 左から3番目（当中の上） - r3: 右端（当小の上） 必ず以下のJSON構造のみを出力してください: { "target_game": 0, "invest": 0, "naki2": 0, "replay": 0, "rep_v": 0, "pick": 0, "v_hit": 0, "sp": 0, "sp_v": 0, "kick": 0, "r10": 0, "r5": 0, "r3": 0 } `; const requestBody = { contents: [{ parts: [ { text: prompt }, { inline_data: { mime_type: mimeType, data: base64Image } } ] }], generationConfig: { response_mime_type: "application/json" } }; const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(requestBody) }); if (!response.ok) { const errorData = await response.json(); throw new Error(errorData.error?.message || 'APIリクエストに失敗しました'); } const resJson = await response.json(); const rawText = resJson.candidates[0].content.parts[0].text; return JSON.parse(rawText); }
